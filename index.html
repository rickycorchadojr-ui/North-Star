<script>

const BIRTH = new Date(1987, 5, 26);

const HORIZON_YEARS = 110;

const MS_DAY = 1000 * 60 * 60 * 24;

 

function startOfDay(d){

  return new Date(d.getFullYear(), d.getMonth(), d.getDate());

}

 

function dayNumber(date){

  return Math.floor((startOfDay(date) - startOfDay(BIRTH)) / MS_DAY) + 1;

}

 

function updateLifeStats(){

  const now = new Date();

  const daysAlive = dayNumber(now);

  const totalDays = Math.floor(365.2425 * HORIZON_YEARS);

  const percent = (daysAlive / totalDays) * 100;

  const ageYears = (now - BIRTH) / (MS_DAY * 365.2425);

 

  document.getElementById("todayText").innerText =

    now.toLocaleDateString(undefined,{weekday:"long",year:"numeric",month:"long",day:"numeric"});

 

  document.getElementById("dayOfLife").innerText = daysAlive.toLocaleString();

  document.getElementById("ageText").innerText = ageYears.toFixed(2);

  document.getElementById("yearOfLife").innerText = Math.ceil(ageYears);

  document.getElementById("lifePercent").innerText = percent.toFixed(2) + "%";

  document.getElementById("lifeProgress").style.width = percent + "%";

}

 

function buildCalendar(){

  const now = new Date();

  const year = now.getFullYear();

  const month = now.getMonth();

 

  document.getElementById("monthLabel").innerText =

    now.toLocaleDateString(undefined,{month:"long",year:"numeric"});

 

  const cal = document.getElementById("calendar");

  cal.innerHTML = "";

 

  const first = new Date(year, month, 1);

  const startDay = first.getDay();

  const daysInMonth = new Date(year, month + 1, 0).getDate();

 

  for(let i=0;i<startDay;i++){

    const blank = document.createElement("div");

    cal.appendChild(blank);

  }

 

  for(let d=1; d<=daysInMonth; d++){

    const date = new Date(year, month, d);

    const cell = document.createElement("div");

    cell.className = "day";

    if(d === now.getDate()) cell.classList.add("today");

 

    cell.innerHTML = `

      <div class="date-num">${d}</div>

      <div class="life-num">Day ${dayNumber(date).toLocaleString()}</div>

    `;

    cal.appendChild(cell);

  }

}

 

function showScreen(id, btn){

  document.querySelectorAll(".screen").forEach(s=>s.classList.remove("active"));

  document.querySelectorAll("button").forEach(b=>b.classList.remove("active"));

  document.getElementById(id).classList.add("active");

  btn.classList.add("active");

}

 

const journal = document.getElementById("journal");

journal.value = localStorage.getItem("northStarJournal_" + dayNumber(new Date())) || "";

journal.addEventListener("input", ()=>{

  localStorage.setItem("northStarJournal_" + dayNumber(new Date()), journal.value);

});

 

updateLifeStats();

buildCalendar();

</script>
